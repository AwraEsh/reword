cmake_minimum_required(VERSION 3.22)
project(Reword VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Sql Concurrent Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HUNSPELL REQUIRED IMPORTED_TARGET hunspell)
qt_add_library(reword_core STATIC src/core/core.cpp src/core/core.h src/core/store.cpp src/core/store.h)
target_link_libraries(reword_core PUBLIC Qt6::Core Qt6::Sql PkgConfig::HUNSPELL)
target_include_directories(reword_core PUBLIC src)
if(BUILD_TESTING)
 qt_add_executable(reword_tests tests/core_test.cpp)
 target_link_libraries(reword_tests PRIVATE reword_core Qt6::Test)
 add_test(NAME core COMMAND reword_tests)
endif()
