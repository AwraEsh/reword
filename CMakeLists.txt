cmake_minimum_required(VERSION 3.22)
project(Reword VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/utf-8)
endif()
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Sql Concurrent Test Network)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
 qt_policy(SET QTP0001 NEW)
 qt_policy(SET QTP0004 NEW)
endif()
if(WIN32)
 find_path(HUNSPELL_INCLUDE_DIR hunspell/hunspell.hxx REQUIRED)
 find_library(HUNSPELL_LIBRARY NAMES hunspell libhunspell hunspell-1.7 libhunspell-1.7 REQUIRED)
 add_library(reword_spell INTERFACE)
 target_include_directories(reword_spell INTERFACE "${HUNSPELL_INCLUDE_DIR}")
 target_link_libraries(reword_spell INTERFACE "${HUNSPELL_LIBRARY}")
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(HUNSPELL REQUIRED IMPORTED_TARGET hunspell)
 add_library(reword_spell INTERFACE)
 target_link_libraries(reword_spell INTERFACE PkgConfig::HUNSPELL)
endif()
qt_add_library(reword_core STATIC src/core/core.cpp src/core/core.h src/core/store.cpp src/core/store.h src/core/private_storage.cpp src/core/private_storage.h)
target_link_libraries(reword_core PUBLIC Qt6::Core Qt6::Sql reword_spell)
target_include_directories(reword_core PUBLIC src)
if(WIN32)
 target_link_libraries(reword_core PRIVATE advapi32)
endif()
if(BUILD_TESTING)
 qt_add_executable(reword_tests tests/core_test.cpp)
 target_link_libraries(reword_tests PRIVATE reword_core Qt6::Test)
 add_test(NAME core COMMAND reword_tests)
endif()

qt_add_library(reword_platform STATIC src/platform/common.cpp src/platform/adapter.h)
target_link_libraries(reword_platform PUBLIC reword_core Qt6::Gui)
if(WIN32)
 target_sources(reword_platform PRIVATE src/platform/windows.cpp)
 target_compile_definitions(reword_platform PRIVATE NOMINMAX UNICODE _UNICODE)
 target_link_libraries(reword_platform PRIVATE uiautomationcore ole32 oleaut32 user32 uuid)
else()
 find_package(Qt6 REQUIRED COMPONENTS DBus)
 pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
 find_package(X11 REQUIRED)
 target_sources(reword_platform PRIVATE src/platform/linux.cpp src/platform/linux_proxy.cpp)
 qt_add_executable(reword-native-helper src/platform/helper_main.cpp)
 target_link_libraries(reword-native-helper PRIVATE reword_platform)
 install(TARGETS reword-native-helper RUNTIME DESTINATION bin)
 target_link_libraries(reword_platform PRIVATE PkgConfig::ATSPI X11::X11 Qt6::DBus)
endif()
qt_add_executable(reword src/app/main.cpp src/app/controller.cpp src/app/controller.h src/app/service.cpp src/app/service.h)
target_link_libraries(reword PRIVATE reword_platform Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Network)
qt_add_qml_module(reword URI Reword VERSION 1.0 RESOURCE_PREFIX / QML_FILES qml/Main.qml qml/SettingRow.qml)
set_target_properties(reword PROPERTIES WIN32_EXECUTABLE TRUE)
if(TARGET reword-native-helper)
 add_dependencies(reword reword-native-helper)
endif()
install(TARGETS reword RUNTIME DESTINATION bin)
install(FILES assets/reword.desktop DESTINATION share/applications)
install(FILES assets/reword.svg DESTINATION share/icons/hicolor/scalable/apps)
option(REWORD_FCITX "Build the Fcitx5 integration" ON)
if(UNIX AND NOT APPLE AND REWORD_FCITX)
 find_package(Fcitx5Core QUIET)
 if(Fcitx5Core_FOUND)
  add_library(reword-fcitx MODULE src/platform/fcitx_addon.cpp src/platform/common.cpp)
  target_link_libraries(reword-fcitx PRIVATE reword_core Fcitx5::Core Qt6::DBus)
  install(TARGETS reword-fcitx LIBRARY DESTINATION lib/${CMAKE_LIBRARY_ARCHITECTURE}/fcitx5)
  install(FILES packaging/reword-fcitx.conf DESTINATION share/fcitx5/addon RENAME reword.conf)
 endif()
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/dictionaries/en_US.dic")
 foreach(app reword reword_tests)
  if(TARGET ${app})
   add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/dictionaries" "$<TARGET_FILE_DIR:${app}>/dictionaries")
  endif()
 endforeach()
 install(DIRECTORY assets/dictionaries/ DESTINATION share/reword/dictionaries)
endif()
if(BUILD_TESTING)
 qt_add_executable(reword_ui_tests tests/ui_test.cpp src/app/controller.cpp src/app/controller.h src/app/service.cpp src/app/service.h)

endif()

 endif()

 qt_add_executable(reword_native_service_probe tests/native_service_probe.cpp src/app/service.cpp src/app/service.h)
